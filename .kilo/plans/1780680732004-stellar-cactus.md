```java
package com.tradej.gateway.broker;
import java.util.*;
public final class DefaultBrokerRegistry implements BrokerRegistry {
    private final Map<BrokerId, BrokerProvider> providers = new LinkedHashMap<>();
    private BrokerId defaultId;

    public synchronized void register(BrokerProvider p) {
        Objects.requireNonNull(p);
        this.providers.put(p.brokerId(), p);
        if (defaultId == null) defaultId = p.brokerId();
    }
    public synchronized void unregister(BrokerId id) {
        providers.remove(id);
        if (Objects.equals(defaultId, id)) {
            defaultId = providers.keySet().stream().findFirst().orElse(null);
        }
    }
    public synchronized Optional<BrokerGateway> gateway(BrokerId id) {
        BrokerProvider p = providers.get(id);
        return p == null ? Optional.empty() : Optional.of(p.gateway());
    }
    public synchronized Optional<BrokerGateway> defaultGateway() {
        return Optional.ofNullable(defaultId).flatMap(this::gateway);
    }
    public synchronized List<BrokerDescriptor> descriptors() {
        return providers.values().stream().map(BrokerProvider::describe).toList();
    }
    public synchronized Optional<BrokerDescriptor> descriptor(BrokerId id) {
        BrokerProvider p = providers.get(id);
        return p == null ? Optional.empty() : Optional.of(p.describe());
    }
}
```

```java
package com.tradej.gateway.broker;
import java.util.*;
public final class ServiceLoaderBrokerRegistry implements BrokerRegistry {
    private final DefaultBrokerRegistry delegate;
    public ServiceLoaderBrokerRegistry() {
        this.delegate = new DefaultBrokerRegistry();
        ServiceLoader.load(BrokerProvider.class).forEach(p -> {
            if (p.isEnabled()) delegate.register(p);
        });
    }
    public ServiceLoaderBrokerRegistry(Collection<BrokerProvider> providers) {
        this.delegate = new DefaultBrokerRegistry();
        providers.stream().filter(BrokerProvider::isEnabled).forEach(delegate::register);
    }
    // delegate
    public synchronized void register(BrokerProvider p) { delegate.register(p); }
    public synchronized void unregister(BrokerId id) { delegate.unregister(id); }
    public synchronized Optional<BrokerGateway> gateway(BrokerId id) { return delegate.gateway(id); }
    public synchronized Optional<BrokerGateway> defaultGateway() { return delegate.defaultGateway(); }
    public synchronized List<BrokerDescriptor> descriptors() { return delegate.descriptors(); }
    public synchronized Optional<BrokerDescriptor> descriptor(BrokerId id) { return delegate.descriptor(id); }
}
```

```java
    public static double pcr(OptionChainSnapshot snapshot) {
        long totalCE = snapshot.strikes().stream().mapToLong(e -> Optional.ofNullable(e.call())
```

### 5.2 DuckDB Engine

**File**: `gateway/src/main/java/com/tradej/gateway/query/DuckDbQueryEngine.java`
```java
package com.tradej.gateway.query;
import java.sql.*;
import java.util.*;
public final class DuckDbQueryEngine implements QueryEngine {
    private final Map<String, MarketDatasource> sources = new HashMap<>();

    public synchronized void registerDatasource(String name, MarketDatasource ds) {
        sources.put(name, ds);
    }

    public QueryResult execute(String sql) {
        long t0 = System.currentTimeMillis();
        try (Connection conn = DriverManager.getConnection("jdbc:duckdb:")) {
            // register datasources as DuckDB tables via JDBC
            for (Map.Entry<String, MarketDatasource> e : sources.entrySet()) {
                PreparedStatement ps = conn.prepareStatement(
                        "CREATE OR REPLACE TABLE " + e.getKey() + " AS SELECT * FROM read_csv_auto('')");
                // implement via JDBC metadata or in-memory registration
            }
            Statement st = conn.createStatement();
            ResultSet rs = st.executeQuery(sql);
            // build rows + columns
            List<Map<String, Object>> rows = new ArrayList<>();
            ResultSetMetaData md = rs.getMetaData();
            List<String> cols = new ArrayList<>();
            for (int i = 1; i <= md.getColumnCount(); i++) cols.add(md.getColumnLabel(i));
            while (rs.next()) {
                Map<String, Object> row = new LinkedHashMap<>();
                for (int i = 1; i <= md.getColumnCount(); i++) row.put(md.getColumnLabel(i), rs.getObject(i));
                rows.add(row);
            }
            return new QueryResult(rows, cols, System.currentTimeMillis() - t0, sql);
        } catch (Exception e) {
            throw new RuntimeException("Query failed: " + e.getMessage(), e);
        }
    }
}
```
